# Road Traffic Contribution Change: Dumbbell Plot (2000–2023)

Horizontal dumbbell plot of the change in road-traffic contribution to the life expectancy gender gap for European countries. Square = 2000, left-pointing triangle = 2023.

```python
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.lines import Line2D
from pathlib import Path

from utils import configure_plot_style, AIBM_COLORS, add_title, add_subtext, add_logo
from fig_utils import get_country_color

configure_plot_style()
```

```python
# Read CSV (generated by extract_road_traffic_changes.py)
csv_path = Path('tables/road_traffic_gap_change_europe.csv')
df = pd.read_csv(csv_path)
df['is_largest_decrease'] = df['is_largest_decrease'].map({'True': True, 'False': False})
# Sort by change (most negative first) to match table order
df = df.sort_values('change').reset_index(drop=True)
df
```

```python
# Horizontal dumbbell: y = country, x = contribution (yr)
# Square at val_2000, left-pointing triangle at val_2023, gray line between
fig, ax = plt.subplots(figsize=(6, 6))

y_pos = range(len(df))
countries = df['country'].values

ax.hlines(y_pos, df['val_2000'], df['val_2023'],
          color=AIBM_COLORS['medium_gray'], linewidth=1, zorder=1)

# Square markers at 2000 (start)
ax.scatter(df['val_2000'], y_pos, marker='s', s=25, color=AIBM_COLORS['light_orange'])

# Left-pointing triangle at 2023 (end)
ax.scatter(df['val_2023'], y_pos, marker='<', s=30, color=AIBM_COLORS['light_orange'])

ax.set_yticks(y_pos)
ax.set_yticklabels(countries, fontsize=10)
ax.invert_yaxis()

# Legend: square = 2000, triangle = 2023
legend_elements = [
    Line2D([0], [0], marker='s', color='w', markerfacecolor=AIBM_COLORS['light_orange'],
           markersize=6, label='2000'),
    Line2D([0], [0], marker='<', color='w', markerfacecolor=AIBM_COLORS['light_orange'],
           markersize=7, label='2023'),
]
ax.legend(handles=legend_elements, loc='lower right')

# Trim 0.5 off top and bottom of y-axis
ax.set_ylim(len(df) - 1.5, 0.5)

ax.set_xlabel('Road traffic contribution to life expectancy gender gap (years)')
add_title('Change in Road Traffic Contribution',
          'European OECD countries (2000-2023)',
          pad=20, x=0, y=1.012)
add_subtext('Source: IHME cause-specific mortality, OWID Life Expectancy.',
            x=0, y=-0.1, align_to_axes=True)
add_logo(filename='logo-hq-small.png', location=(1.1, -0.12), align_to_axes=True)

ax.set_xlim(0, 1.8)
ax.set_xticks([0.0, 0.5, 1.0, 1.5])
for x in [0.1, 0.5, 1.0, 1.5]:
    ax.axvline(x=x, color=AIBM_COLORS['light_gray'], alpha=0.8, linewidth=1, zorder=0)

plt.tight_layout()
output_path = Path('figs/road_traffic_dumbbell_europe.png')
output_path.parent.mkdir(exist_ok=True)
plt.savefig(output_path, dpi=150, bbox_inches='tight')
plt.show()
print(f"Saved: {output_path}")
```
